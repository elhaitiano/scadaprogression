# ================= Agenda =================

class Evento(models.Model):
    """Reunião, visita/intervenção num sítio ou prazo. As tarefas do planeamento aparecem
    na agenda a partir do próprio modelo Tarefa (não são copiadas para aqui)."""

    class Tipo(models.TextChoices):
        REUNIAO = "reuniao", "Reunião"
        VISITA = "visita", "Visita / intervenção"
        PRAZO = "prazo", "Prazo / marco"

    class Estado(models.TextChoices):
        PREVISTO = "previsto", "Previsto"
        REALIZADO = "realizado", "Realizado"
        CANCELADO = "cancelado", "Cancelado"

    tipo = models.CharField(max_length=8, choices=Tipo.choices)
    titulo = models.CharField("Título", max_length=160)
    data = models.DateField()
    data_fim = models.DateField("Até", null=True, blank=True, help_text="Só para eventos de vários dias")
    hora_inicio = models.TimeField("Hora de início", null=True, blank=True, help_text="Vazio = dia inteiro")
    hora_fim = models.TimeField("Hora de fim", null=True, blank=True)
    local = models.CharField(max_length=160, blank=True, help_text="Ex.: CD Marçal, sala de reuniões EPAL, Teams")
    sitio = models.ForeignKey(Sitio, null=True, blank=True, on_delete=models.SET_NULL, related_name="eventos")
    tarefa = models.ForeignKey(Tarefa, null=True, blank=True, on_delete=models.SET_NULL, related_name="eventos")
    participantes = models.ManyToManyField(Pessoa, blank=True, related_name="eventos")
    participantes_externos = models.CharField("Participantes externos", max_length=200, blank=True,
                                              help_text="Ex.: EPAL (Eng. X), SUEZ Digital Solutions")
    descricao = models.TextField("Descrição / ordem de trabalhos", blank=True)
    estado = models.CharField(max_length=10, choices=Estado.choices, default=Estado.PREVISTO)
    criado_por = models.ForeignKey(settings.AUTH_USER_MODEL, null=True, blank=True,
                                   on_delete=models.SET_NULL, related_name="+")
    criado_em = models.DateTimeField(auto_now_add=True)
    actualizado_em = models.DateTimeField(auto_now=True)
    actualizado_por = models.ForeignKey(settings.AUTH_USER_MODEL, null=True, blank=True,
                                        on_delete=models.SET_NULL, related_name="+")

    class Meta:
        ordering = ["data", "hora_inicio", "id"]
        verbose_name = "evento"
        constraints = [
            models.CheckConstraint(condition=models.Q(data_fim__isnull=True) | models.Q(data_fim__gte=models.F("data")),
                                   name="evento_fim_depois_inicio"),
        ]

    def __str__(self):
        return f"{self.data:%d/%m/%Y} · {self.titulo}"
